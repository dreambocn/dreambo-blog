#### 日志消费Flume启停脚本
```bash
#!/bin/bash

case $1 in
    "start")
        echo " --------启动 server3 日志数据flume-------"

        # 检查 HDFS 是否处于安全模式
        SAFE_MODE_STATUS=$(ssh server3 "source /etc/profile; hdfs dfsadmin -safemode get")
        if [[ $SAFE_MODE_STATUS == *"ON"* ]]; then
            echo "HDFS 处于安全模式，正在关闭..."
            ssh server3 "source /etc/profile; hdfs dfsadmin -safemode leave"
        else
            echo "HDFS 未处于安全模式，直接启动 Flume"
        fi

        ssh server3 "source /etc/profile; nohup /opt/module/flume/bin/flume-ng agent -n a1 -c /opt/module/flume/conf -f /opt/module/flume/job/kafka_to_hdfs_log.conf >/dev/null 2>&1 &"
        ;;
    "stop")
        echo " --------停止 server3 日志数据flume-------"
        ssh server3 "PIDS=\$(ps -ef | grep kafka_to_hdfs_log | grep -v grep | awk '{print \$2}'); if [ -n \"\$PIDS\" ]; then echo \"Killing PIDs: \$PIDS\"; echo \$PIDS | xargs -n1 kill -9; else echo \"No kafka_to_hdfs_log process found.\"; fi"
        ;;
esac
```
